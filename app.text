VEHICLE_CLASSES = ['police', 'ambulance', 'bus', 'car', 'motorcycle', 'truck']
MIN_AREA = 141524
MAX_AREA = 141525

# Payment prices
PRICES = {
    'car': 10000,
    'truck': 40000,
    'bus': 30000,
    'motorcycle': 5000,
    'police': 0,

}

# Utilities

def normalize_plate(text):
    return ''.join(ch for ch in text.upper() if ch.isalnum())

def extract_plate_text(frame, x1, y1, x2, y2):
    plate_crop = frame[y1:y2, x1:x2]
    if plate_crop.size == 0:
        return "Unknown"
    results = reader.readtext(plate_crop)
    text = ""
    for (_, detected_text, conf) in results:
        if conf > 0.4:
            text += detected_text + " "
    return normalize_plate(text.strip()) if text else "Unknown"

def save_to_db(image, vehicle_type, plate, captured_time):
    if plate == "Unknown":
        return
    sql = "INSERT INTO captures (image, car_type, plate, captured_time, status, payment_method) VALUES (%s, %s, %s, %s, %s, %s)"
    val = (image, vehicle_type, plate, captured_time, 'Unpaid', 'Not Paid')
    cursor.execute(sql, val)
    db.commit()
    print(f"✅ Saved {vehicle_type} with plate {plate}")

def get_video_fps():
    fps = cap.get(cv2.CAP_PROP_FPS)
    return fps if fps > 0 else 30.0

def get_stats():
    """Get statistics for dashboard with PHP-compatible structure"""
    try:
        # Total cars
        cursor.execute("SELECT COUNT(*) as total_cars FROM captures")
        total_cars = cursor.fetchone()['total_cars']
        
        # Paid cars (status starts with 'Paid')
        cursor.execute("SELECT COUNT(*) as paid_cars FROM captures WHERE status LIKE 'Paid%'")
        paid_cars = cursor.fetchone()['paid_cars']
        
        # Unpaid cars
        cursor.execute("SELECT COUNT(*) as unpaid_cars FROM captures WHERE status = 'Unpaid'")
        unpaid_cars = cursor.fetchone()['unpaid_cars']
        
        # Payment methods
        cursor.execute("SELECT COUNT(*) as momo_payments FROM captures WHERE payment_method = 'MOMO'")
        momo_payments = cursor.fetchone()['momo_payments']
        
        cursor.execute("SELECT COUNT(*) as cash_payments FROM captures WHERE payment_method = 'CASH'")
        cash_payments = cursor.fetchone()['cash_payments']
        
        # Calculate total revenue from all paid vehicles
        cursor.execute("SELECT status FROM captures WHERE status LIKE 'Paid%'")
        paid_rows = cursor.fetchall()
        
        revenue = 0
        for row in paid_rows:
            # Extract numbers from status like "Paid FRW 40,000"
            if row['status'] and row['status'].startswith('Paid'):
                numbers = re.findall(r'\d+', row['status'].replace(',', ''))
                if numbers:
                    revenue += int(numbers[0])
        
        print(f"Stats calculated: total_cars={total_cars}, paid_cars={paid_cars}, unpaid_cars={unpaid_cars}, momo_payments={momo_payments}, cash_payments={cash_payments}, revenue={revenue}")
        
        return {
            'total_cars': total_cars,
            'paid_cars': paid_cars,
            'unpaid_cars': unpaid_cars,
            'momo_payments': momo_payments,
            'cash_payments': cash_payments,
            'revenue': revenue
        }
    except Exception as e:
        print(f"Error getting stats: {e}")
        return {
            'total_cars': 0,
            'paid_cars': 0,
            'unpaid_cars': 0,
            'momo_payments': 0,
            'cash_payments': 0,
            'revenue': 0
        }

# -----------------------------
# Frame Generator
# -----------------------------
def gen_frames():
    video_fps = get_video_fps()
    frame_interval = 1.0 / video_fps
    
    while True:
        start_time = time.time()
        success, frame = cap.read()
        if not success:
            break

        results = vehicle_model.predict(frame, conf=0.4, verbose=False)
        annotated = frame.copy()

        for box in results[0].boxes:
            cls_id = int(box.cls[0])
            label = vehicle_model.names[cls_id].lower()

            if label in VEHICLE_CLASSES:
                x1, y1, x2, y2 = map(int, box.xyxy[0])
                width, height = x2 - x1, y2 - y1
                area = width * height

                if MIN_AREA <= area <= MAX_AREA:
                    vehicle_crop = frame[y1:y2, x1:x2]
                    plate_results = plate_model.predict(vehicle_crop, conf=0.5, verbose=False)
                    plate_text = "Unknown"

                    for pbox in plate_results[0].boxes:
                        px1, py1, px2, py2 = map(int, pbox.xyxy[0])
                        plate_text = extract_plate_text(vehicle_crop, px1, py1, px2, py2)
                        cv2.rectangle(vehicle_crop, (px1, py1), (px2, py2), (0, 255, 255), 2)
                        cv2.putText(vehicle_crop, plate_text, (px1, py1 - 5),
                                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 255), 2)

                    timestamp = datetime.now()
                    formatted_time = timestamp.strftime("%Y-%m-%d %H:%M:%S")
                    filename = f"static/captures/{label}_{timestamp.strftime('%Y%m%d_%H%M%S')}.jpg"
                    cv2.imwrite(filename, vehicle_crop)
                    save_to_db(filename, label.title(), plate_text, formatted_time)

                    cv2.rectangle(annotated, (x1, y1), (x2, y2), (0, 255, 0), 3)
                    cv2.putText(annotated, f"{label.title()} - {plate_text}",
                                (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
                else:
                    cv2.rectangle(annotated, (x1, y1), (x2, y2), (0, 0, 255), 2)
                    cv2.putText(annotated, f"{label.title()} - REJECTED",
                                (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)

        _, buffer = cv2.imencode('.jpg', annotated)
        yield (b'--frame\r\n'
               b'Content-Type: image/jpeg\r\n\r\n' + buffer.tobytes() + b'\r\n')
        
        time.sleep(max(0, frame_interval - (time.time() - start_time)))

@app.route('/process_payment', methods=['POST'])
def process_payment():
    """Handle MOMO and CASH payments"""
    try:
        data = request.get_json()
        vehicle_id = data.get('id')
        method = data.get('method')  # 'MOMO' or 'CASH'
        amount = data.get('amount')
        
        status_text = f"Paid FRW {amount:,}"
        
        cursor.execute("UPDATE captures SET status = %s, payment_method = %s WHERE id = %s", 
                     (status_text, method, vehicle_id))
        db.commit()
        
        return jsonify({'success': True, 'status': status_text})
    except Exception as e:
        print(f"Payment error: {e}")
        return jsonify({'success': False, 'error': str(e)})

@app.route('/delete_vehicle', methods=['POST'])
def delete_vehicle():
    """Delete a vehicle record"""
    try:
        data = request.get_json()
        vehicle_id = data.get('id')
        
        if not vehicle_id:
            return jsonify({'success': False, 'message': 'Vehicle ID is required'})
        
        print(f"Attempting to delete vehicle ID: {vehicle_id}")
        
        # First, get the image path to delete the file
        cursor.execute("SELECT image FROM captures WHERE id = %s", (vehicle_id,))
        result = cursor.fetchone()
        
        if result:
            image_path = result['image']
            
            # Delete the image file from filesystem
            try:
                if os.path.exists(image_path):
                    os.remove(image_path)
                    print(f"✅ Deleted image file: {image_path}")
                else:
                    print(f"⚠️ Image file not found: {image_path}")
            except Exception as e:
                print(f"⚠️ Could not delete image file: {e}")
            
            # Delete the record from database
            cursor.execute("DELETE FROM captures WHERE id = %s", (vehicle_id,))
            db.commit()
            
            deleted_rows = cursor.rowcount
            if deleted_rows > 0:
                print(f"✅ Deleted vehicle record ID: {vehicle_id}")
                return jsonify({'success': True, 'message': 'Record deleted successfully'})
            else:
                return jsonify({'success': False, 'message': 'Record not found in database'})
        else:
            return jsonify({'success': False, 'message': 'Record not found'})
            
    except Exception as e:
        print(f"❌ Delete error: {e}")
        db.rollback()
        return jsonify({'success': False, 'message': f'Database error: {str(e)}'})

@app.route('/get_analytics')
def get_analytics():
    """Get analytics data for history page"""
    try:
        # Calculate payment method counts
        cursor.execute("SELECT COUNT(*) as momo_payments FROM captures WHERE payment_method = 'MOMO'")
        momo_payments = cursor.fetchone()['momo_payments']
        
        cursor.execute("SELECT COUNT(*) as cash_payments FROM captures WHERE payment_method = 'CASH'")
        cash_payments = cursor.fetchone()['cash_payments']
        
        cursor.execute("SELECT COUNT(*) as unpaid_cars FROM captures WHERE status = 'Unpaid'")
        unpaid_cars = cursor.fetchone()['unpaid_cars']
        
        # Calculate revenue by car type
        cursor.execute("""
            SELECT car_type, 
                   SUM(CASE WHEN status LIKE 'Paid%' THEN 
                       CASE 
                           WHEN car_type = 'Car' THEN 10000
                           WHEN car_type = 'Truck' THEN 40000
                           WHEN car_type = 'Bus' THEN 30000
                           WHEN car_type = 'Motorcycle' THEN 5000
                           ELSE 10000
                       END
                   ELSE 0 END) as revenue
            FROM captures 
            GROUP BY car_type
        """)
        revenue_by_type = cursor.fetchall()
        
        revenue_data = []
        revenue_labels = []
        for row in revenue_by_type:
            revenue_data.append(row['revenue'] or 0)
            revenue_labels.append(row['car_type'])
        
        # Calculate total potential and actual revenue
        cursor.execute("""
            SELECT 
                SUM(CASE 
                    WHEN car_type = 'Car' THEN 10000
                    WHEN car_type = 'Truck' THEN 40000
                    WHEN car_type = 'Bus' THEN 30000
                    WHEN car_type = 'Motorcycle' THEN 5000
                    ELSE 10000
                END) as total_potential
            FROM captures
        """)
        total_potential = cursor.fetchone()['total_potential'] or 0
        
        cursor.execute("""
            SELECT 
                SUM(CASE WHEN status LIKE 'Paid%' THEN 
                    CASE 
                        WHEN car_type = 'Car' THEN 10000
                        WHEN car_type = 'Truck' THEN 40000
                        WHEN car_type = 'Bus' THEN 30000
                        WHEN car_type = 'Motorcycle' THEN 5000
                        ELSE 10000
                    END
                ELSE 0 END) as total_actual
            FROM captures
        """)
        total_actual = cursor.fetchone()['total_actual'] or 0
        
        total_loss = total_potential - total_actual
        
        return jsonify({
            'momo_payments': momo_payments,
            'cash_payments': cash_payments,
            'unpaid_cars': unpaid_cars,
            'revenue_data': revenue_data,
            'revenue_labels': revenue_labels,
            'total_potential': total_potential,
            'total_actual': total_actual,
            'total_loss': total_loss
        })
    except Exception as e:
        print(f"Analytics error: {e}")
        return jsonify({
            'momo_payments': 0,
            'cash_payments': 0,
            'unpaid_cars': 0,
            'revenue_data': [],
            'revenue_labels': [],
            'total_potential': 0,
            'total_actual': 0,
            'total_loss': 0
        })

def init_database():

            CREATE TABLE IF NOT EXISTS captures (
                id INT AUTO_INCREMENT PRIMARY KEY,
                image VARCHAR(255) NOT NULL,
                car_type VARCHAR(50),
                plate VARCHAR(50),
                captured_time DATETIME DEFAULT CURRENT_TIMESTAMP,
                status VARCHAR(50) DEFAULT 'Unpaid',
                payment_method VARCHAR(50) DEFAULT 'Not Paid'
            )
        """)
        
        # Check if we have any data
        cursor.execute("SELECT COUNT(*) as count FROM captures")
        count = cursor.fetchone()['count']
        
        if count == 0:
            # Add some test data
            test_data = [
